cmake_minimum_required(VERSION 3.20)
project(rudy_routeatlas VERSION 0.1.0 LANGUAGES CXX)
option(ROUTEATLAS_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /WX /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Werror)
  if(ROUTEATLAS_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(routeatlas_core src/prefix.cpp)
target_sources(routeatlas_core PRIVATE src/index.cpp)
